3.3 Prepare data

3.3.1 Population identifier

Population identity was defined using study identity and rounded geographic coordinates, with latitude and longitude rounded to two decimal places prior to constructing the population identifier.

Code 
mixed_pat <- mixed_pat_raw %>%
  mutate(
    Lat_round = round(Lat, 2),
    Long_round = round(Long, 2),
    population_id = paste(Study_ID, Lat_round, Long_round, sep = "_"),
    population_id = as.factor(population_id)
  )

3.3.2 Study and species formatting

The following step formats the main grouping variables used in the models and calculates the total number of broods by combining mixed- and non-mixed-brood counts.

Code 
mixed_pat <- mixed_pat %>%
  mutate(
    Study_ID = as.factor(Study_ID),
    species_phylo_brms = as.factor(Clements_name),
    species_nonphylo = as.factor(Clements_name),

    n_mixed_broods = as.numeric(n_mixed_broods),
    n_non_mixed_broods = as.numeric(n_non_mixed_broods),
    n_total_broods = n_mixed_broods + n_non_mixed_broods
  )

3.3.3 Phylogeny preparation

Species names were harmonised between the mixed-paternity dataset and the phylogenetic tree prior to pruning the tree and generating the phylogenetic covariance matrix used in the Bayesian models.

Code 
species_data <- mixed_pat %>%

  as.character() %>%
  unique()

species_tree <- tree$tip.label

phylo_mat <- phylo_mat[
  sort(rownames(phylo_mat)),
  sort(colnames(phylo_mat))
]

mixed_pat <- mixed_pat %>%
  filter(species_phylo_brms %in% rownames(phylo_mat)) %>%
  mutate(
    species_phylo_brms = droplevels(species_phylo_brms),
    species_nonphylo = droplevels(species_nonphylo)
  )

3.4.2 Common scaling of continuous moderators

The centring and scaling constants are calculated once from the 619-record overall analytical dataset, before moderator-specific complete-case filtering. The same constants are then reused in every single-moderator and multivariable analysis.

Code 
data_overall_study <- data_overall_study %>%
  mutate(
    log_body_mass = replace(
      body_mass,
      !is.na(body_mass) & body_mass <= 0,
      NA_real_
    ),
    log_body_mass = log10(log_body_mass)
  )

scaling_constants <- data_overall_study %>%
  summarise(
    study_duration_mean = mean(study_duration, na.rm = TRUE),
    study_duration_sd = sd(study_duration, na.rm = TRUE),
    CS_max_mean = mean(CS_max, na.rm = TRUE),
    CS_max_sd = sd(CS_max, na.rm = TRUE),
    log_body_mass_mean = mean(log_body_mass, na.rm = TRUE),
    log_body_mass_sd = sd(log_body_mass, na.rm = TRUE),
    abs_lat_mean = mean(abs_lat, na.rm = TRUE),
    abs_lat_sd = sd(abs_lat, na.rm = TRUE)
  )

data_overall_study <- data_overall_study %>%
  mutate(
    study_duration_sc =
      (study_duration - scaling_constants$study_duration_mean) /
      scaling_constants$study_duration_sd,
    CS_max_sc =
      (CS_max - scaling_constants$CS_max_mean) /
      scaling_constants$CS_max_sd,
    log_body_mass_sc =
      (log_body_mass - scaling_constants$log_body_mass_mean) /
      scaling_constants$log_body_mass_sd,
    abs_lat_sc =
      (abs_lat - scaling_constants$abs_lat_mean) /
      scaling_constants$abs_lat_sd
  )

scaling_table <- tibble::tribble(
  ~Predictor, ~N, ~Mean, ~SD, ~Unit,
  "Study duration", sum(!is.na(data_overall_study$study_duration)),
    scaling_constants$study_duration_mean, scaling_constants$study_duration_sd, "years",
  "Maximum clutch size", sum(!is.na(data_overall_study$CS_max)),
    scaling_constants$CS_max_mean, scaling_constants$CS_max_sd, "eggs",
  "Body mass", sum(!is.na(data_overall_study$log_body_mass)),
    scaling_constants$log_body_mass_mean, scaling_constants$log_body_mass_sd, "log10(g)",
  "Absolute latitude", sum(!is.na(data_overall_study$abs_lat)),
    scaling_constants$abs_lat_mean, scaling_constants$abs_lat_sd, "degrees"
) %>%
  mutate(across(c(Mean, SD), ~ round(.x, 3)))

## Relationship among candidate random-effects grouping factors

Before fitting the models, we evaluated the degree of overlap and nesting among candidate random-effects grouping factors, including species identity, study identity, and the pragmatic population identifier.

```{r}
#| label: random-effect-structure
#| echo: false
#| message: false
#| warning: false

count_one_match <- function(data, group_var, match_var, comparison_label) {
  data %>%
    distinct({{ group_var }}, {{ match_var }}) %>%
    count({{ group_var }}, name = "n_matches") %>%
    summarise(
      Comparison = comparison_label,
      Groups_with_only_one_match = sum(n_matches == 1),
      Total_groups = n(),
      Percent_only_one_match = round(mean(n_matches == 1) * 100, 1),
      .groups = "drop"
    )
}

random_effect_structure <- bind_rows(
  count_one_match(data_overall_study, species_nonphylo, population_id, "Populations per species"),
  count_one_match(data_overall_study, population_id, species_nonphylo, "Species per population"),
  count_one_match(data_overall_study, Study_ID, population_id, "Populations per reference"),
  count_one_match(data_overall_study, population_id, Study_ID, "References per population"),
  count_one_match(data_overall_study, species_nonphylo, Study_ID, "References per species"),
  count_one_match(data_overall_study, Study_ID, species_nonphylo, "Species per reference")
)
```
